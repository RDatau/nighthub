const formatSpeed = (bytesPerSec?: number) => {
  if (!bytesPerSec || bytesPerSec === 0) return '';
  if (bytesPerSec < 1024) return `${Math.round(bytesPerSec)} B/s`;
  if (bytesPerSec < 1024 * 1024) return `${(bytesPerSec / 1024).toFixed(1)} KB/s`;
  return `${(bytesPerSec / (1024 * 1024)).toFixed(2)} MB/s`;
};

const getHostname = (urlString: string) => {
  if (!urlString) return '';
  let cleanUrl = urlString.trim();
  if (!/^https?:\/\//i.test(cleanUrl)) {
    cleanUrl = 'http://' + cleanUrl;
  }
  try {
    const parsed = new URL(cleanUrl);
    return parsed.hostname;
  } catch (e) {
    return urlString;
  }
};

const isMatchingHostname = (h1: string, h2: string) => {
  if (!h1 || !h2) return false;
  const cleanHost1 = h1.toLowerCase().replace(/^www\./, '');
  const cleanHost2 = h2.toLowerCase().replace(/^www\./, '');
  if (cleanHost1 === cleanHost2) return true;

  // helper to get apex/registrable domain
  const getApex = (h: string) => {
    const parts = h.split('.');
    if (parts.length > 2) {
      // Check if it has a common two-part TLD (e.g. co.uk, com.br, etc)
      const secondLevelDomains = ['com', 'co', 'net', 'org', 'gov', 'edu', 'mil', 'ac', 'or', 'sch', 'web', 'my'];
      if (secondLevelDomains.includes(parts[parts.length - 2]) && parts.length > 3) {
        return parts.slice(-3).join('.');
      }
      return parts.slice(-2).join('.');
    }
    return h;
  };

  const apex1 = getApex(cleanHost1);
  const apex2 = getApex(cleanHost2);
  return apex1 === apex2;
};

                return s;

  const handleReorder = async (draggedId: string, targetId: string, position: 'before' | 'after' | 'inside') => {
    const draggedBm = bookmarks.find(b => b.id === draggedId);
    const targetBm = bookmarks.find(b => b.id === targetId);

    if (position === 'inside') {
      siblings.push({ ...draggedBm, parentId: newParentId });
    } else {
      if (position === 'after') {
        targetIndex += 1;
      }
      siblings.splice(targetIndex, 0, { ...draggedBm, parentId: newParentId });

          next[idx] = ub;

      });
      return next;
    });

    for (const ub of updatedBookmarks) {
      await dbService.saveBookmark(ub);
    }
  };

  const handleImportBookmarks = () => {
    const input = document.createElement('input');
    input.type = 'file';
    input.accept = '.html';
    input.onchange = async (e) => {
      const file = (e.target as HTMLInputElement).files?.[0];
      if (!file) return;
      const text = await file.text();
      const parser = new DOMParser();
      const doc = parser.parseFromString(text, 'text/html');
      
      const newBookmarks: Bookmark[] = [];
      
      const parseNode = (node: Element, parentId: string | null) => {
        const children = Array.from(node.children);
        for (let i = 0; i < children.length; i++) {
          const child = children[i];
          if (child.tagName.toLowerCase() === 'dt') {
            const h3 = Array.from(child.children).find(c => c.tagName.toLowerCase() === 'h3');
            const a = Array.from(child.children).find(c => c.tagName.toLowerCase() === 'a');
            
            if (h3) {
              const folderId = `bm-${Date.now()}-${Math.random().toString(36).substr(2, 9)}`;
              newBookmarks.push({
                id: folderId,
                title: h3.textContent || 'New Folder',
                type: 'folder',
                parentId,
                createdAt: parseInt(h3.getAttribute('ADD_DATE') || '0') * 1000 || Date.now()
              });
              
              // Find the next sibling that is a DL
              let nextSibling = child.nextElementSibling;
              let parsedDl = false;
              while (nextSibling && nextSibling.tagName.toLowerCase() !== 'dt') {
                if (nextSibling.tagName.toLowerCase() === 'dl') {
                  parseNode(nextSibling, folderId);
                  parsedDl = true;
                  break;
                }
                nextSibling = nextSibling.nextElementSibling;
              }
              
              if (!parsedDl) {
                // Sometimes DL is inside DT
                const innerDl = Array.from(child.children).find(c => c.tagName.toLowerCase() === 'dl');
                if (innerDl) {
                  parseNode(innerDl, folderId);
                }
              }
            } else if (a) {
              newBookmarks.push({
                id: `bm-${Date.now()}-${Math.random().toString(36).substr(2, 9)}`,
                title: a.textContent || a.getAttribute('href') || '',
                url: a.getAttribute('href') || '',
                type: 'link',
                parentId,
                createdAt: parseInt(a.getAttribute('ADD_DATE') || '0') * 1000 || Date.now()
              });
            }
          } else if (child.tagName.toLowerCase() === 'dl') {
             parseNode(child, parentId);
          }
        }
      };
      
      const rootDl = doc.querySelector('dl');
      if (rootDl) {
        parseNode(rootDl, null);
      }
      
      for (const bm of newBookmarks) {
        await dbService.saveBookmark(bm);
      }
      setBookmarks(prev => [...prev, ...newBookmarks]);
      setShowBookmarkMenu(false);
    };
    input.click();
  };

        const ipRegex = /^(\d{1,3}\.){3}\d{1,3}(:\d+)?(\/.*)?$/;
        const localhostRegex = /^localhost(:\d+)?(\/.*)?$/;
        return domainRegex.test(str) || ipRegex.test(str) || localhostRegex.test(str);
      }

      }

  };

  // Synchronize suggestions with Electron transparent frameless window
  useEffect(() => {
    const ipc = getIpcRenderer();
    if (!ipc) return;

    const updateSuggestionsMenu = () => {
      const isVisible = showUrlSuggestions && show && url.trim() !== '';
      if (isVisible) {
        const addressBar = addressBarRef.current;
        if (addressBar) {
          const rect = addressBar.getBoundingClientRect();
          ipc.send('show-suggestions-menu', {
            bounds: {
              x: rect.x,
              y: rect.y,
              width: rect.width,
              height: rect.height
            },
            url,
            searchEngine,
            webHistory,
            lang
          });
        }
      } else {
        ipc.send('hide-suggestions-menu');
      }
    };

    updateSuggestionsMenu();

    window.addEventListener('resize', updateSuggestionsMenu);
    return () => {
      window.removeEventListener('resize', updateSuggestionsMenu);
    };
  }, [showUrlSuggestions, url, searchEngine, webHistory, lang, show, getIpcRenderer]);

  // Handle suggestion selection from the frameless suggestions window
  useEffect(() => {
    const ipc = getIpcRenderer();
    if (!ipc) return;

    const handleSelectSuggestion = (event: any, targetUrl: string) => {
      setUrl(targetUrl);
      setShowUrlSuggestions(false);
      setTimeout(() => {
        startSniffing(targetUrl);
      }, 50);
    };

    ipc.on('select-suggestion', handleSelectSuggestion);
    return () => {
      ipc.off('select-suggestion', handleSelectSuggestion);
    };
  }, [getIpcRenderer, startSniffing]);

  // Listen for 'password-submitted' event from Electron main process
  useEffect(() => {
    const ipc = getIpcRenderer();
    if (!ipc) return;

    const handlePasswordSubmitted = async (_event: any, data: { url: string; username: string; password: string }) => {
      if (!data || !data.password) return;

      const hostname = getHostname(data.url);
      const allCreds = await dbService.getCredentials();
      const existing = allCreds.find(c => isMatchingHostname(getHostname(c.url), hostname) && c.username === data.username);

      if (existing) {
        setIsUpdatePrompt(true);
      } else {
        setIsUpdatePrompt(false);
      }

      setPendingCredential(data);
    };

    ipc.on('password-submitted', handlePasswordSubmitted);
    return () => {
      ipc.off('password-submitted', handlePasswordSubmitted);
    };
  }, [getIpcRenderer]);

  // Focus address bar whenever a blank tab is active or created
  useEffect(() => {
    if (!show) return;
    const currentActiveTab = tabs.find(t => t.id === activeTabId) || tabsRef.current.find(t => t.id === activeTabId);
    const isBlank = !currentActiveTab || !currentActiveTab.url || currentActiveTab.url.trim() === '' || currentActiveTab.url === 'about:blank';

    if (isBlank) {
      isProgrammaticFocusRef.current = true;
      const timer = setTimeout(() => {
        isProgrammaticFocusRef.current = true;
        addressBarRef.current?.focus();
        if (currentActiveTab?.url) {
          addressBarRef.current?.select();
        }
      }, 50);

      const ipc = getIpcRenderer();
      if (ipc) {
        ipc.send('focus-main-window');
      }
      return () => clearTimeout(timer);
    } else {
      if (addressBarRef.current) {
        addressBarRef.current.blur();
      }
      setShowUrlSuggestions(false);
    }
  }, [show, activeTabId, tabs]);

  const handleNextTab = useCallback(() => {
    const currentTabs = tabsRef.current;
    if (currentTabs.length === 0) return;
    const currentActiveId = activeTabIdRef.current;
    let currentIndex = currentTabs.findIndex(t => t.id === currentActiveId);
    if (currentIndex === -1) currentIndex = 0;
    const nextIndex = (currentIndex + 1) % currentTabs.length;
    handleTabClick(currentTabs[nextIndex].id);
  }, [handleTabClick]);

  const handlePrevTab = useCallback(() => {
    const currentTabs = tabsRef.current;
    if (currentTabs.length === 0) return;
    const currentActiveId = activeTabIdRef.current;
    let currentIndex = currentTabs.findIndex(t => t.id === currentActiveId);
    if (currentIndex === -1) currentIndex = 0;
    const prevIndex = (currentIndex - 1 + currentTabs.length) % currentTabs.length;
    handleTabClick(currentTabs[prevIndex].id);
  }, [handleTabClick]);

  const tabShortcutsRef = useRef({
    handleNextTab,
    handlePrevTab,
    handleDuplicateTab,
    handleTabAdd,
    handleTabClose,
    handleAddBookmark,
    showHistorySidebar,
    setShowHistorySidebar,
    setShowLeftSidebar,
    setShowSidebar,
    setShowDownloadSidebar,
    setShowPasswordsSidebar,
  });

  useEffect(() => {
    tabShortcutsRef.current = {
      handleNextTab,
      handlePrevTab,
      handleDuplicateTab,
      handleTabAdd,
      handleTabClose,
      handleAddBookmark,
      showHistorySidebar,
      setShowHistorySidebar,
      setShowLeftSidebar,
      setShowSidebar,
      setShowDownloadSidebar,
      setShowPasswordsSidebar,
    };
  });

  // Keyboard shortcuts (Ctrl+T, Ctrl+W, Ctrl+Shift+K, Ctrl+Tab, etc.)
  useEffect(() => {
    if (!show) return;

    const handleKeyDown = (e: KeyboardEvent) => {
      // Tab management shortcuts must ALWAYS work, even when cursor is active inside the Address bar or search inputs
      
      // 1. Shortcut Ctrl+Shift+K to duplicate active tab
      if ((e.ctrlKey || e.metaKey) && e.shiftKey && e.key.toLowerCase() === 'k') {
        e.preventDefault();
        e.stopPropagation();
        tabShortcutsRef.current.handleDuplicateTab(activeTabIdRef.current);
        return;
      }

      // 2. Shortcut Ctrl+T to create new blank tab and focus Address bar
      if ((e.ctrlKey || e.metaKey) && !e.shiftKey && e.key.toLowerCase() === 't') {
        e.preventDefault();
        e.stopPropagation();
        tabShortcutsRef.current.handleTabAdd();
        return;
      }

      // 3. Shortcut Ctrl+W to close active tab
      if ((e.ctrlKey || e.metaKey) && !e.shiftKey && e.key.toLowerCase() === 'w') {
        e.preventDefault();
        e.stopPropagation();
        tabShortcutsRef.current.handleTabClose(activeTabIdRef.current, e);
        return;
      }

      // 4. Shortcut Ctrl+Tab / Ctrl+Shift+Tab to switch tabs
      if ((e.ctrlKey || e.metaKey) && (e.key === 'Tab' || e.code === 'Tab')) {
        e.preventDefault();
        e.stopPropagation();
        if (e.shiftKey) {
          tabShortcutsRef.current.handlePrevTab();
        } else {
          tabShortcutsRef.current.handleNextTab();
        }
        return;
      }

      if ((e.ctrlKey || e.metaKey) && e.key.toLowerCase() === 'f') {
        e.preventDefault();
        const ipc = getIpcRenderer();
        if (ipc) {
          ipc.send('show-find-view');
        }
        return;
      }

      if ((e.ctrlKey || e.metaKey) && e.key.toLowerCase() === 'd') {
        e.preventDefault();
        tabShortcutsRef.current.handleAddBookmark();
        return;
      }

      if ((e.ctrlKey || e.metaKey) && e.key.toLowerCase() === 'h') {
        e.preventDefault();
        const newState = !tabShortcutsRef.current.showHistorySidebar;
        tabShortcutsRef.current.setShowHistorySidebar(newState);
        if (newState) {
          tabShortcutsRef.current.setShowLeftSidebar(false);
          tabShortcutsRef.current.setShowSidebar(false);
          tabShortcutsRef.current.setShowDownloadSidebar(false);
          tabShortcutsRef.current.setShowPasswordsSidebar(false);
        }
        return;
      }

      // Single-key shortcuts should NOT trigger when user is typing in inputs or textareas
      if (e.target instanceof HTMLInputElement || e.target instanceof HTMLTextAreaElement) return;

      if (e.key.toLowerCase() === 'f') {
        e.preventDefault();
        const ipc = getIpcRenderer();
        if (ipc) {
          ipc.send('sniffer-toggle-video-fullscreen', activeTabIdRef.current);
        }
        return;
      }
    };
    
    window.addEventListener('keydown', handleKeyDown, true);
    
    const ipc = getIpcRenderer();
    
    const onIpcCloseTab = (e: any, tabId?: string) => {
      const targetId = tabId || activeTabIdRef.current;
      if (targetId) tabShortcutsRef.current.handleTabClose(targetId);
    };
    const onIpcNewTab = () => tabShortcutsRef.current.handleTabAdd();
    const onIpcDuplicateTab = (e: any, tabId?: string, overrideUrl?: string) => {
      const targetId = tabId || activeTabIdRef.current;
      tabShortcutsRef.current.handleDuplicateTab(targetId, overrideUrl);
    };
    const onIpcNextTab = () => tabShortcutsRef.current.handleNextTab();
    const onIpcPrevTab = () => tabShortcutsRef.current.handlePrevTab();

    if (ipc) {
      ipc.on('sniffer-shortcut-close-tab', onIpcCloseTab);
      ipc.on('sniffer-shortcut-new-tab', onIpcNewTab);
      ipc.on('sniffer-shortcut-duplicate-tab', onIpcDuplicateTab);
      ipc.on('sniffer-shortcut-next-tab', onIpcNextTab);
      ipc.on('sniffer-shortcut-prev-tab', onIpcPrevTab);
    }

    return () => {
      window.removeEventListener('keydown', handleKeyDown, true);
      if (ipc) {
        ipc.removeListener('sniffer-shortcut-close-tab', onIpcCloseTab);
        ipc.removeListener('sniffer-shortcut-new-tab', onIpcNewTab);
        ipc.removeListener('sniffer-shortcut-duplicate-tab', onIpcDuplicateTab);
        ipc.removeListener('sniffer-shortcut-next-tab', onIpcNextTab);
        ipc.removeListener('sniffer-shortcut-prev-tab', onIpcPrevTab);
      }
    };
  }, [show, getIpcRenderer]);

  const rootBookmarks = useMemo(() => {
    return bookmarks.filter(b => !b.parentId).sort((a, b) => {
      if (bookmarkSort === 'manual') {
        const orderA = a.order !== undefined ? a.order : 999999;
        const orderB = b.order !== undefined ? b.order : 999999;
        if (orderA !== orderB) return orderA - orderB;
        return b.createdAt - a.createdAt;
      }
      
      // Folders always first
      if (a.type === 'folder' && b.type !== 'folder') return -1;
      if (a.type !== 'folder' && b.type === 'folder') return 1;
      
      if (bookmarkSort === 'a-z') {
        return a.title.localeCompare(b.title);
      } else {
        return b.createdAt - a.createdAt; // Newest first
      }
    });
  }, [bookmarks, bookmarkSort]);

  const activeTab = useMemo(() => tabs.find(t => t.id === activeTabId), [tabs, activeTabId]);
  const canGoBack = activeTab?.canGoBack || false;
  const canGoForward = activeTab?.canGoForward || false;

  return (
    <div className={`absolute inset-0 z-[100] bg-transparent dark:bg-transparent flex flex-col overflow-hidden ${isResizing ? 'select-none' : ''} ${!show ? 'hidden' : ''}`}>
      {!isFullscreen && (
        <div className="flex-none bg-transparent flex flex-col relative z-50">
          <TitleBar 
            isSnifferOpen={true}
            tabs={tabs}
            activeTabId={activeTabId}
            onTabClick={handleTabClick}
            onTabClose={handleTabClose}
            onTabAdd={handleTabAdd}
            onTabMuteToggle={handleTabMuteToggle}
            onTabReorder={handleTabReorder}
            onTabContextMenu={(id, isActive) => {
              const ipc = getIpcRenderer();
              if (ipc) ipc.send('show-tab-context-menu', { tabId: id, isActive });
            }}
            onTitleBarContextMenu={() => {
              const ipc = getIpcRenderer();
              if (ipc) ipc.send('show-titlebar-context-menu');
            }}
            tabPreviews={tabPreviews}
            lang={lang}
          />
          <div className="flex flex-col bg-white/60 dark:bg-zinc-700/40 backdrop-blur-md relative z-10">
            <div className="p-2 flex items-center gap-2 relative">

            {/* Progress Bar Removed */}
            
            <div className="flex items-center gap-1">
            <button 
              onClick={goBack}
              disabled={!isSniffing || !canGoBack}
              className="p-2 bg-transparent hover:bg-black/10 dark:hover:bg-white/10 disabled:opacity-20 disabled:hover:bg-transparent disabled:hover:text-zinc-700 dark:disabled:hover:text-zinc-300 disabled:cursor-default text-zinc-700 dark:text-zinc-300 hover:text-zinc-900 dark:hover:text-white rounded-lg transition-colors"
              title="Back"
            >
              <ArrowLeft size={16} />
            </button>
            <button 
              onClick={goForward}
              disabled={!isSniffing || !canGoForward}
              className="p-2 bg-transparent hover:bg-black/10 dark:hover:bg-white/10 disabled:opacity-20 disabled:hover:bg-transparent disabled:hover:text-zinc-700 dark:disabled:hover:text-zinc-300 disabled:cursor-default text-zinc-700 dark:text-zinc-300 hover:text-zinc-900 dark:hover:text-white rounded-lg transition-colors"
              title="Forward"
            >
              <ArrowRight size={16} />
            </button>
            <button 
              onClick={reloadSniffer}
              disabled={!isSniffing}
              className="p-2 bg-transparent hover:bg-black/10 dark:hover:bg-white/10 disabled:opacity-30 disabled:hover:bg-transparent text-zinc-700 dark:text-zinc-300 hover:text-zinc-900 dark:hover:text-white rounded-lg transition-colors"
              title="Reload"
            >
              <RefreshCw size={16} />
            </button>
            <button
              onClick={(e) => {
                if (e.ctrlKey) {
                  handleTabAdd(homeUrl);
                } else {
                  startSniffing(homeUrl);
                }
              }}
              onContextMenu={(e) => {
                e.preventDefault();
                const ipc = getIpcRenderer();
                if (ipc) {
                  ipc.send('show-home-context-menu');
                }
              }}
              className="p-2 bg-transparent hover:bg-black/10 dark:hover:bg-white/10 text-zinc-700 dark:text-zinc-300 hover:text-zinc-900 dark:hover:text-white rounded-lg transition-colors"
              title="Home (Ctrl+Click to open in new tab)"
            >
              <Home size={16} />
            </button>
          </div>

          <div className="flex-1 relative flex items-center">
            <div className="absolute left-4 text-zinc-500">
              {activeTab?.url === 'nighthub://settings' || pageFavicon === settingsFavicon ? (
                <img src={nighthubFavicon} alt="favicon" className="w-4.5 h-4.5 object-contain rounded-sm" />
              ) : (pageFavicon && !failedFavicons[pageFavicon]) ? (
                <img 
                  src={pageFavicon} 
                  alt="favicon" 
                  onError={() => {
                    if (pageFavicon) {
                      setFailedFavicons(prev => ({ ...prev, [pageFavicon]: true }));
                    }
                  }}
                  className="w-4 h-4 object-contain" 
                />
              ) : (
                <Globe size={14} />
              )}
            </div>
            <input 
              ref={addressBarRef}
              type="text" 
              value={url}
              onChange={(e) => {
                setUrl(e.target.value);
                setShowUrlSuggestions(true);
              }}
              onMouseDown={() => {
                isMouseDownRef.current = true;
              }}
              onFocus={(e) => {
                const isProgrammatic = isProgrammaticFocusRef.current;
                const isClick = isMouseDownRef.current;
                const isBlank = !url || url.trim() === '' || url === 'about:blank';
                
                isProgrammaticFocusRef.current = false;
                isMouseDownRef.current = false;
                
                if (isProgrammatic || isClick || isBlank) {
                  addressBarJustFocusedRef.current = true;
                  if (url && url.trim() !== '') {
                    e.target.select();
                    setShowUrlSuggestions(true);
                  }
                } else {
                  // Automatic focus restoration (e.g. context menu dismiss or adblock blur).
                  // Blur the input to prevent focus stealing and selection, and restore focus to active view.
                  e.target.blur();
                  const ipc = getIpcRenderer();
                  if (ipc) ipc.send('focus-active-view');
                }
              }}
              onClick={(e) => {
                if (addressBarJustFocusedRef.current) {
                  e.currentTarget.select();
                  addressBarJustFocusedRef.current = false;
                }
                setShowUrlSuggestions(true);
              }}
              onBlur={(e) => {
                addressBarJustFocusedRef.current = false;
                // Keep dropdown open if clicking/focusing inside suggestions dropdown
                const currentTarget = e.currentTarget;
                setTimeout(() => {
                  if (
                    urlSuggestionsRef.current && 
                    urlSuggestionsRef.current.contains(document.activeElement)
                  ) {
                    return;
                  }
                  setShowUrlSuggestions(false);
                }, 200);
              }}
              onKeyDown={(e) => {
                if (e.key === 'Enter') {
                  setShowUrlSuggestions(false);
                  if (e.ctrlKey) {
                    let formattedUrl = url.trim();
                    if (!formattedUrl.includes('.')) {
                      formattedUrl = `${formattedUrl}.com`;
                    }
                    startSniffing(formattedUrl);
                  } else {
                    startSniffing();
                  }
                } else if (e.key === 'Escape') {
                  setShowUrlSuggestions(false);
                }
              }}
              placeholder={`Search ${searchEngine.charAt(0).toUpperCase() + searchEngine.slice(1)} or type a URL`}
              className="w-full bg-white dark:bg-zinc-700/50 text-zinc-700 dark:text-zinc-300 text-sm pl-10 pr-36 py-2 rounded-full border border-zinc-400/30 dark:border-white/5 focus:border-zinc-500/40 dark:focus:border-zinc-500/40 focus:outline-none transition-all"
            />
            
            <div className="absolute right-3 flex items-center gap-1">
              <div className="relative">
                <button
                  ref={keyButtonRef}
                  onClick={() => {
                    const ipc = getIpcRenderer();
                    if (ipc && keyButtonRef.current) {
                      const rect = keyButtonRef.current.getBoundingClientRect();
                      ipc.send('show-passwords-menu', {
                        bounds: {
                          x: rect.x,
                          y: rect.y,
                          width: rect.width,
                          height: rect.height
                        },
                        url,
                        lang
                      });
                    } else {
                      setShowPasswordsDropdown(!showPasswordsDropdown);
                    }
                  }}
                  className={`p-1 transition-colors ${currentSiteCredentials.length > 0 ? 'text-amber-500 hover:text-amber-600' : 'text-zinc-500 hover:text-orange-600 dark:hover:text-amber-500'}`}
                  title={lang === 'id' ? 'Sandi Tersimpan di Situs Ini' : 'Saved Passwords on This Site'}
                >
                  <Key size={14} className={currentSiteCredentials.length > 0 ? 'fill-amber-500/20' : ''} />
                </button>
                {showPasswordsDropdown && (
                  <div className="absolute right-0 mt-2 w-72 bg-white dark:bg-[#1a1a1a] rounded-xl shadow-2xl border border-black/10 dark:border-white/10 p-3 z-[2000] flex flex-col gap-2 text-left">
                    <div className="flex items-center justify-between border-b border-black/5 dark:border-white/5 pb-1.5">
                      <span className="text-xs font-extrabold text-zinc-900 dark:text-white flex items-center gap-1.5">
                        <Key size={12} className="text-amber-500" />
                        {lang === 'id' ? 'Sandi Situs Ini' : 'Saved Passwords'}
                      </span>
                      <button 
                        onClick={() => {
                          setShowPasswordsDropdown(false);
                          setShowPasswordsSidebar(true);
                          setShowLeftSidebar(false);
                          setShowSidebar(false);
                          setShowDownloadSidebar(false);
                          setShowHistorySidebar(false);
                          setShowAdblockSidebar(false);
                        }}
                        className="text-[10px] font-bold text-orange-500 dark:text-amber-500 hover:underline"
                      >
                        {lang === 'id' ? 'Kelola' : 'Manage'}
                      </button>
                    </div>
                    {currentSiteCredentials.length === 0 ? (
                      <div className="text-center py-4 text-xs text-zinc-500">
                        {lang === 'id' ? 'Tidak ada sandi tersimpan untuk situs ini.' : 'No saved passwords for this site.'}
                      </div>
                    ) : (
                      <div className="flex flex-col gap-2 max-h-48 overflow-y-auto custom-scrollbar">
                        {currentSiteCredentials.map((cred) => (
                          <div key={cred.id} className="p-2 rounded-xl bg-black/5 dark:bg-white/5 flex flex-col gap-1 text-xs">
                            <div className="flex items-center justify-between font-medium text-zinc-800 dark:text-zinc-200">
                              <span className="truncate max-w-[120px]">{cred.username}</span>
                              <div className="flex items-center gap-1">
                                <button
                                  onClick={() => {
                                    navigator.clipboard.writeText(cred.username);
                                  }}
                                  className="p-1 text-zinc-400 hover:text-zinc-600 dark:hover:text-zinc-200"
                                  title={lang === 'id' ? 'Salin Username' : 'Copy Username'}
                                >
                                  <Copy size={10} />
                                </button>
                                <button
                                  onClick={() => {
                                    navigator.clipboard.writeText(cred.password);
                                  }}
                                  className="p-1 text-zinc-400 hover:text-zinc-600 dark:hover:text-zinc-200"
                                  title={lang === 'id' ? 'Salin Sandi' : 'Copy Password'}
                                >
                                  <Copy size={10} />
                                </button>
                              </div>
                            </div>
                            <button
                              onClick={() => {
                                const ipc = getIpcRenderer();
                                if (ipc) {
                                  ipc.send('sniffer-autofill', { username: cred.username, password: cred.password });
                                }
                                setShowPasswordsDropdown(false);
                              }}
                              className="w-full py-1 text-[10px] font-extrabold uppercase tracking-widest text-center bg-orange-500 hover:bg-orange-600 dark:bg-amber-500 dark:hover:bg-amber-500 text-white dark:text-black rounded-lg transition-colors"
                            >
                              {lang === 'id' ? 'Isi Otomatis' : 'Autofill'}
                            </button>
                          </div>
                        ))}
                      </div>
                    )}
                  </div>
                )}
              </div>
              <button 
                onClick={() => {
                  const isBookmarked = bookmarks.some(b => b.url === url);
                  if (isBookmarked) {
                    const bm = bookmarks.find(b => b.url === url);
                    if (bm) handleDeleteBookmark(bm.id);
                  } else {
                    handleAddBookmark();
                  }
                }}
                className="p-1 text-zinc-500 hover:text-orange-600 dark:hover:text-amber-500 transition-colors"
                title={t('addBookmarkTooltip', lang)}
              >
                <Star size={14} className={bookmarks.some(b => b.url === url) ? 'fill-orange-500 text-orange-500 dark:fill-amber-500 dark:text-amber-500' : ''} />
              </button>
              <button 
                onClick={() => startSniffing()}
                className="p-1 text-zinc-500 hover:text-orange-600 dark:hover:text-amber-500 transition-colors"
                title="Search/Go"
              >
                <Search size={14} />
              </button>
            </div>

            {/* Overlay Update / Save Password */}
            {pendingCredential && (
              <div className="absolute right-0 top-full mt-2 z-[10000] bg-white dark:bg-[#1a1a1a] rounded-xl shadow-2xl w-80 overflow-hidden border border-black/10 dark:border-white/10 animate-in fade-in slide-in-from-top-1 duration-150 text-left">
                <div className="p-3 border-b border-black/10 dark:border-white/10 flex items-center gap-2">
                  <div className="w-7 h-7 rounded-lg bg-amber-500/15 flex items-center justify-center shrink-0">
                    <Key className="text-amber-500" size={14} />
                  </div>
                  <div>
                    <h3 className="text-xs font-bold text-zinc-900 dark:text-white">
                      {isUpdatePrompt 
                        ? (lang === 'id' ? 'Perbarui Sandi?' : 'Update Password?') 
                        : (lang === 'id' ? 'Simpan Sandi?' : 'Save Password?')}
                    </h3>
                    <p className="text-[10px] text-zinc-500">
                      NightHUB Password Manager
                    </p>
                  </div>
                </div>

                <div className="p-3 flex flex-col gap-2 text-xs">
                  {/* Domain */}
                  <div>
                    <label className="text-[9px] font-bold uppercase tracking-wider text-zinc-400 block mb-0.5">Situs</label>
                    <div className="px-2 py-1 bg-black/5 dark:bg-white/5 border border-black/5 dark:border-white/5 rounded-md text-[11px] text-zinc-500 select-all truncate font-medium">
                      {getHostname(pendingCredential.url)}
                    </div>
                  </div>

                  {/* Username */}
                  <div>
                    <label className="text-[9px] font-bold uppercase tracking-wider text-zinc-400 block mb-0.5">
                      {lang === 'id' ? 'Nama Pengguna' : 'Username'}
                    </label>
                    <input
                      type="text"
                      value={pendingCredential.username}
                      onChange={(e) => setPendingCredential(prev => prev ? { ...prev, username: e.target.value } : null)}
                      className="w-full px-2 py-1 bg-black/5 dark:bg-white/5 border border-black/10 dark:border-white/10 rounded-md text-[11px] text-zinc-900 dark:text-white focus:outline-none focus:border-orange-500/50 dark:focus:border-amber-500/50 transition-colors font-medium"
                    />
                  </div>

                  {/* Password */}
                  <div>
                    <label className="text-[9px] font-bold uppercase tracking-wider text-zinc-400 block mb-0.5">

                          className={`w-full text-left px-3 py-2 text-sm flex items-center gap-2 transition-colors ${bookmarkSort === 'date' ? 'text-orange-600 dark:text-amber-500 bg-orange-50 dark:bg-amber-500/10' : 'text-zinc-700 dark:text-zinc-300 hover:bg-black/5 dark:hover:bg-white/5 hover:text-zinc-900 dark:hover:text-white'}`}
                        >
                          <Clock size={14} />
                          {t('sortByDateModified', lang)}
                        </button>
                        <div className="h-px bg-black/10 dark:bg-white/10 my-1" />
                        <button
                          onClick={() => {
                            setFolderPrompt({ show: true, mode: 'create' });
                            setFolderPromptInput('');
                            setShowBookmarkMenu(false);
                          }}
                          className="w-full text-left px-3 py-2 text-sm text-zinc-700 dark:text-zinc-300 hover:bg-black/5 dark:hover:bg-white/5 hover:text-zinc-900 dark:hover:text-white flex items-center gap-2 transition-colors"
                        >
                          <FolderPlus size={14} />
                          {t('newFolder', lang)}
                        </button>
                        <div className="h-px bg-black/10 dark:bg-white/10 my-1" />
                        <button
                          onClick={() => {
                            handleExportBookmarks();
                            setShowBookmarkMenu(false);
                          }}
                          className="w-full text-left px-3 py-2 text-sm text-zinc-700 dark:text-zinc-300 hover:bg-black/5 dark:hover:bg-white/5 hover:text-zinc-900 dark:hover:text-white flex items-center gap-2 transition-colors"
                        >
                          <FileDown size={14} />
                          {t('exportHtmlFile', lang)}
                        </button>
                        <button
                          onClick={() => {
                            handleImportBookmarks();
                            setShowBookmarkMenu(false);
                          }}
                          className="w-full text-left px-3 py-2 text-sm text-zinc-700 dark:text-zinc-300 hover:bg-black/5 dark:hover:bg-white/5 hover:text-zinc-900 dark:hover:text-white flex items-center gap-2 transition-colors"
                        >
                          <FileUp size={14} />
                          {t('importHtmlFile', lang)}
                        </button>
                        <div className="h-px bg-black/10 dark:bg-white/10 my-1" />
                        <button
                          onClick={() => {
                            setDeleteConfirm({ show: true, mode: 'all' });
                            setShowBookmarkMenu(false);
                          }}
                          className="w-full text-left px-3 py-2 text-sm text-red-600 dark:text-red-400 hover:bg-red-50 dark:hover:bg-red-500/10 hover:text-red-700 dark:hover:text-red-300 flex items-center gap-2 transition-colors"
                        >
                          <Trash2 size={14} />
                          {t('deleteAllBookmarks', lang)}
                        </button>
                      </div>
                    </>
                  )}
                </div>
                <button 
                  onClick={() => setShowLeftSidebar(false)}
                  className="p-1.5 text-zinc-500 hover:text-orange-600 dark:hover:text-amber-500 hover:bg-orange-500/10 dark:hover:bg-amber-500/10 rounded-lg transition-colors"
                  title="Close Bookmark Manager"
                >
                  <X size={16} />
                </button>
              </div>
            </div>
            
            <div 
              className="flex-1 overflow-y-auto p-2 custom-scrollbar"
              onContextMenu={(e) => {
                e.preventDefault();
                setBookmarkBgContextMenu({ x: e.clientX, y: e.clientY });
              }}
              onDragOver={(e) => {
                e.preventDefault();
              }}
              onDrop={async (e) => {
                e.preventDefault();
                const data = e.dataTransfer.getData('application/json');
                if (data) {
                  try {
                    const droppedBm = JSON.parse(data) as Bookmark;
                    if (droppedBm.parentId !== null) {
                      const updatedBm = { ...droppedBm, parentId: null };
                      await dbService.saveBookmark(updatedBm);
                      setBookmarks(prev => prev.map(b => b.id === updatedBm.id ? updatedBm : b));
                    }
                    return;
                  } catch (err) {}
                }
                
                let droppedUrl = e.dataTransfer.getData('text/plain');
                if (droppedUrl) {
                  droppedUrl = droppedUrl.trim();
                  if (!droppedUrl.startsWith('http://') && !droppedUrl.startsWith('https://')) {
                    if (droppedUrl.includes('.')) {
                      droppedUrl = `https://${droppedUrl}`;
                    } else {
                      return;
                    }
                  }
                  
                  // Resolve title from tabs
                  let resolvedTitle = '';
                  const cleanUrl = (u: string) => u.replace(/^https?:\/\//i, '').replace(/\/$/, '').toLowerCase();
                  const targetClean = cleanUrl(droppedUrl);
                  const matchedTab = tabs.find(t => cleanUrl(t.url) === targetClean);
                  if (matchedTab && matchedTab.title && matchedTab.title !== 'New Tab' && matchedTab.title !== 'Loading...') {
                    resolvedTitle = matchedTab.title;
                  } else {
                    resolvedTitle = getHostname(droppedUrl) || droppedUrl;
                  }

                  const newBm: Bookmark = {
                    id: `bm-${Date.now()}`,
                    title: resolvedTitle,
                    url: droppedUrl,
                    type: 'link',
                    parentId: null,
                    createdAt: Date.now()
                  };
                  await dbService.saveBookmark(newBm);
                  setBookmarks(prev => [...prev, newBm]);
                }

            >
              {rootBookmarks.length === 0 ? (

                  <BookmarkIcon size={32} className="mb-2 opacity-50" />
                  <p className="text-sm">Kosong.</p>
                </div>

                <div className="space-y-1">

                      bookmarks={bookmarks}
                      level={0}
                      sortBy={bookmarkSort}

                      onOpenNewTab={(url, background) => {
                        handleTabAdd(url, background);
                      }}
                      onContextMenu={(e, bookmark) => {
                        setBookmarkContextMenu({
                          x: e.clientX,
                          y: e.clientY,
                          bookmark
                        });
                      }}
                      onUpdate={async (updatedBm) => {

                        setBookmarks(prev => {
                          const exists = prev.some(b => b.id === updatedBm.id);
                          if (exists) {
                            return prev.map(b => b.id === updatedBm.id ? updatedBm : b);
                          } else {
                            return [...prev, updatedBm];
                          }
                        });
                      }}
                      onDelete={async (deletedId) => {
                        const allDeletedIds = new Set<string>();
                        const collectDeletedIds = (parentId: string) => {
                          const children = bookmarks.filter(b => b.parentId === parentId);
                          for (const child of children) {
                            allDeletedIds.add(child.id);
                            if (child.type === 'folder') collectDeletedIds(child.id);
                          }
                        };
                        collectDeletedIds(deletedId);
                        allDeletedIds.add(deletedId);
                        
                        for (const id of allDeletedIds) {
                          await dbService.deleteBookmark(id);
                        }
                        
                        setBookmarks(prev => prev.filter(b => !allDeletedIds.has(b.id)));
                      }}
                      onReorder={handleReorder}
                      isDescendant={isDescendant}
                    />
                  ))}
                </div>
              )}
            </div>
          </div>
        )}


        {/* Sidebar (Video Terdeteksi) */}
        {showSidebar && !isFullscreen && (
          <div 
            style={{ width: sidebarWidth }}
            className="flex-none bg-white/5 dark:bg-black/40 backdrop-blur-md border border-transparent dark:border-transparent rounded-2xl flex flex-col z-10 h-full overflow-hidden shadow-sm"
          >
            <div className="flex items-center justify-between h-14 px-4 bg-zinc-200/5 dark:bg-white/5 border-b border-black/10 dark:border-white/10 flex-none">
              <h2 className="text-sm font-semibold text-zinc-900 dark:text-white flex items-center gap-2">
                <ListVideo size={16} className="text-orange-500 dark:text-amber-500" />
                {t('detectedVideosTitle', lang, { count: detectedVideos.length })}
              </h2>
              <div className="flex items-center gap-2">
                <button 
                  onClick={() => setDetectedVideos([])}
                  className="text-[10px] text-zinc-500 hover:text-red-400 transition-colors uppercase tracking-widest"
                >
                  {t('clear', lang)}
                </button>
                <button 
                  onClick={() => setShowSidebar(false)}
                  className="p-1.5 text-zinc-500 hover:text-orange-600 dark:hover:text-amber-500 hover:bg-orange-500/10 dark:hover:bg-amber-500/10 rounded-lg transition-colors"
                  title="Close Video List"
                >
                  <X size={16} />
                </button>
              </div>
            </div>
            
            <div className="flex-1 overflow-y-auto p-4 space-y-3 custom-scrollbar min-h-0">
              {detectedVideos.length === 0 ? (
                <div className="h-full flex flex-col items-center justify-center text-zinc-700 dark:text-zinc-300 gap-4 opacity-50">
                  <Search size={48} strokeWidth={1} />
                  <p className="text-xs font-mono uppercase tracking-widest text-center px-4">
                    {isSniffing ? 'Memantau jaringan...' : 'Mulai jelajahi untuk menangkap video'}
                  </p>
                </div>
              ) : (
                detectedVideos.map((video, idx) => (
                  <div 
                    key={idx}
                    className="group bg-black/5 dark:bg-white/5 border border-black/5 dark:border-white/5 hover:border-orange-500/30 dark:hover:border-amber-500/30 dark:border-amber-500/30 rounded-xl p-3 transition-all duration-300 flex flex-col gap-2"
                  >
                    <div className="flex items-center gap-2 flex-wrap">
                      <span className="px-1.5 py-0.5 bg-orange-500/20 dark:bg-amber-500/20 text-orange-600 dark:text-amber-500 text-[9px] font-bold rounded uppercase">
                        {video.type}
                      </span>
                      {video.size && (
                        <span className="px-1.5 py-0.5 bg-blue-500/20 text-blue-400 text-[9px] font-bold rounded uppercase">
                          {video.size}
                        </span>
                      )}
                      <span className="text-[9px] text-zinc-500 font-mono ml-auto">
                        {new Date(video.timestamp).toLocaleTimeString()}
                      </span>
                    </div>
                    <div className="flex flex-col gap-0.5 min-w-0">
                      <h3 
                        className="text-zinc-900 dark:text-white text-xs font-medium truncate select-text cursor-pointer hover:text-orange-400 dark:hover:text-amber-400 transition-colors" 
                        title={video.title || video.url}
                        onClick={() => handleCopy(video.url)}
                      >
                        {video.title || video.url}
                      </h3>
                      {video.title && (
                        <span className="text-[10px] text-zinc-500 truncate select-text">
                          {video.url}
                        </span>
                      )}
                    </div>
                    
                    <div className="flex gap-2 mt-1">
                      <button 
                        onClick={() => handleCopy(video.url)}
                        className="flex-1 py-1.5 bg-black/5 dark:bg-white/5 hover:bg-black/10 dark:hover:bg-white/10 text-zinc-900 dark:text-white rounded-lg transition-all flex justify-center items-center"
                        title={t('copyUrl', lang)}
                      >
                        {copiedUrl === video.url ? <Check size={14} className="text-emerald-500" /> : <Copy size={14} />}
                      </button>
                      <button 
                        onClick={() => onPlay(video.url, url, video.title || pageTitle)}
                        className="flex-1 py-1.5 bg-orange-500/20 dark:bg-amber-500/20 hover:bg-orange-500/30 dark:hover:bg-amber-500/30 text-orange-600 dark:text-amber-500 rounded-lg transition-all flex justify-center items-center"
                        title={t('directPlay', lang)}
                      >
                        <Play size={14} fill="currentColor" />
                      </button>
                      <button 
                        onClick={() => handleDownload(video)}
                        className={`flex-1 py-1.5 rounded-lg transition-all flex justify-center items-center ${
                          downloads[video.url]?.status === 'downloading' || downloads[video.url]?.status === 'starting'
                            ? 'bg-blue-500/10 text-blue-300 cursor-not-allowed'
                            : downloads[video.url]?.status === 'completed'
                            ? 'bg-emerald-500/20 text-emerald-500'
                            : 'bg-blue-500/20 hover:bg-blue-500/30 text-blue-400'
                        }`}
                        title={t('downloadVideo', lang)}
                        disabled={downloads[video.url]?.status === 'downloading' || downloads[video.url]?.status === 'starting'}
                      >
                        {downloads[video.url]?.status === 'downloading' || downloads[video.url]?.status === 'starting' ? (
                          <RefreshCw size={14} className="animate-spin" />
                        ) : downloads[video.url]?.status === 'completed' ? (
                          <Check size={14} />
                        ) : (
                          <Download size={14} />
                        )}
                      </button>
                      <a 
                        href={video.url}
                        target="_blank"
                        rel="noopener noreferrer"
                        className="flex-1 py-1.5 bg-black/5 dark:bg-white/5 hover:bg-black/10 dark:hover:bg-white/10 text-zinc-900 dark:text-white rounded-lg transition-all flex justify-center items-center"
                        title={t('openInExternalBrowser', lang)}
                      >
                        <ExternalLink size={14} />
                      </a>
                    </div>

                    {/* Download Progress UI */}
                    {(downloads[video.url]?.status === 'downloading' || downloads[video.url]?.status === 'starting') && (
                      <div className="mt-2 space-y-1.5">
                        <div className="flex justify-between items-center text-[10px] font-mono">
                          <div className="flex items-center gap-2">
                            <span className="text-blue-400 uppercase tracking-widest">
                              {downloads[video.url]?.status === 'starting' ? t('downloadStarting', lang) : t('downloading', lang)}
                            </span>
                            {downloads[video.url]?.speed !== undefined ? (
                              <span className="text-emerald-500 font-medium">
                                {formatSpeed(downloads[video.url].speed)}
                              </span>
                            ) : null}
                          </div>
                          <div className="flex items-center gap-2 text-zinc-700 dark:text-zinc-300">
                            {downloads[video.url]?.timemark && (
                              <span>{downloads[video.url].timemark}</span>
                            )}
                            {downloads[video.url]?.percent > 0 && (
                              <span className="text-blue-400 font-bold">{Math.round(downloads[video.url].percent)}%</span>
                            )}
                          </div>
                        </div>
                        <div className="flex items-center gap-2">
                          <div className="h-1.5 flex-1 bg-black/5 dark:bg-black/50 rounded-full overflow-hidden border border-black/5 dark:border-white/5">
                            <div 
                              className="h-full bg-gradient-to-r from-blue-600 to-blue-400 rounded-full transition-all duration-300 relative"
                              style={{ width: `${Math.max(5, downloads[video.url]?.percent || 0)}%` }}
                            >
                              <div className="absolute inset-0 bg-black/20 dark:bg-white/20 w-full animate-[shimmer_1s_infinite] -skew-x-12"></div>
                            </div>
                          </div>
                          <button
                            onClick={() => handleCancelDownload(video.url)}
                            className="text-zinc-500 hover:text-red-500 transition-colors p-1 rounded-md hover:bg-red-500/10 flex-shrink-0"
                            title={t('cancelDownload', lang)}
                          >
                            <X size={14} strokeWidth={3} />
                          </button>
                        </div>
                      </div>
                    )}

                    {downloads[video.url]?.status === 'error' && (
                      <div className="text-[10px] text-red-500 mt-1 bg-red-500/10 p-1.5 rounded border border-red-500/20">
                        {t('downloadFailed', lang)}: {downloads[video.url].error}
                      </div>
                    )}
                    
                    {downloads[video.url]?.status === 'cancelled' && (
                      <div className="text-[10px] text-orange-600 dark:text-amber-500 mt-1 bg-orange-500/10 dark:bg-amber-500/10 p-1.5 rounded border border-orange-500/20 dark:border-amber-500/20">
                        {t('downloadCancelled', lang)}
                      </div>
                    )}
                  </div>
                ))
              )}
            </div>
          </div>
        )}

        {/* Download Manager Sidebar */}
        <DownloadManagerSidebar 
          show={showDownloadSidebar && !isFullscreen} 
          onClose={() => setShowDownloadSidebar(false)} 
          width={sidebarWidth} 
        />

        {/* History Sidebar */}
        {showHistorySidebar && !isFullscreen && (
          <div 
            style={{ width: sidebarWidth }}
            className="flex-none bg-white/5 dark:bg-black/40 backdrop-blur-md border border-transparent dark:border-transparent rounded-2xl flex flex-col z-10 h-full overflow-hidden shadow-sm"
          >
            <div className="flex items-center justify-between h-14 px-4 bg-zinc-200/5 dark:bg-white/5 border-b border-black/10 dark:border-white/10 flex-none">
              <h2 className="text-sm font-semibold text-zinc-900 dark:text-white flex items-center gap-2">
                <History size={16} className="text-orange-500 dark:text-amber-500" />
                History
              </h2>
              <button 
                onClick={() => setShowHistorySidebar(false)}
                className="p-1.5 text-zinc-500 hover:text-orange-600 dark:hover:text-amber-500 hover:bg-orange-500/10 dark:hover:bg-amber-500/10 rounded-lg transition-colors"
                title="Close History"
              >
                <X size={16} />
              </button>
            </div>
            
            <div className="flex p-1 bg-black/5 dark:bg-white/5 mx-2 mt-2 rounded-xl border border-black/5 dark:border-white/5">
              <button
                onClick={() => setHistoryTab('web')}
                className={`flex-1 py-1.5 text-[11px] font-bold uppercase tracking-wider rounded-lg transition-all ${historyTab === 'web' ? 'bg-white dark:bg-zinc-800 text-orange-600 dark:text-amber-500 shadow-sm' : 'text-zinc-500 hover:text-zinc-700 dark:hover:text-zinc-300'}`}
              >
                {t('historyTabWeb', lang)}
              </button>
              <button
                onClick={() => setHistoryTab('video')}
                className={`flex-1 py-1.5 text-[11px] font-bold uppercase tracking-wider rounded-lg transition-all ${historyTab === 'video' ? 'bg-white dark:bg-zinc-800 text-orange-600 dark:text-amber-500 shadow-sm' : 'text-zinc-500 hover:text-zinc-700 dark:hover:text-zinc-300'}`}
              >
                {t('historyTabVideo', lang)}
              </button>
            </div>

            <div className="flex-1 overflow-y-auto p-2 custom-scrollbar">
              {historyTab === 'web' && (
                <>
                  <div className="flex justify-end mb-2">
                    <button
                      onClick={async () => {
                        await dbService.clearWebHistory();
                        setWebHistory([]);
                      }}
                      className="text-[10px] text-zinc-500 hover:text-red-400 transition-colors uppercase tracking-widest"
                    >
                      {t('clearAllWebHistory', lang)}
                    </button>
                  </div>
                  {webHistory.length === 0 ? (
                    <div className="h-full flex flex-col items-center justify-center text-zinc-500 gap-2 opacity-50">
                      <Globe size={32} />
                      <p className="text-sm">{t('noWebHistory', lang)}</p>
                    </div>
                  ) : (
                    <div className="flex flex-col gap-1">
                      {webHistory.map((item) => (
                        <div key={item.id} className="group flex items-center gap-2 p-2 rounded-xl hover:bg-black/5 dark:hover:bg-white/5 transition-colors">
                          <div className="w-6 h-6 rounded-lg bg-black/5 dark:bg-white/5 flex items-center justify-center shrink-0">
                            {(item.favicon && !failedFavicons[item.favicon]) ? (
                              <img 
                                src={item.favicon} 
                                alt="" 
                                onError={() => {
                                  if (item.favicon) {
                                    setFailedFavicons(prev => ({ ...prev, [item.favicon]: true }));
                                  }
                                }}
                                className="w-4 h-4 object-contain" 
                              />
                            ) : (
                              <Globe size={12} className="text-zinc-500" />
                            )}
                          </div>
                          <div className="flex-1 min-w-0 cursor-pointer" onClick={() => startSniffing(item.url)}>
                            <div className="text-sm text-zinc-900 dark:text-white truncate font-medium">{item.title}</div>
                            <div className="text-[10px] text-zinc-500 truncate">{item.url}</div>
                          </div>
                          <button
                            onClick={async (e) => {
                              e.stopPropagation();
                              await dbService.deleteWebHistory(item.id);
                              setWebHistory(prev => prev.filter(h => h.id !== item.id));
                            }}
                            className="p-1.5 text-zinc-500 hover:text-red-500 hover:bg-red-500/10 rounded-lg transition-colors opacity-0 group-hover:opacity-100"
                            title={t('delete', lang)}
                          >
                            <Trash2 size={14} />
                          </button>
                        </div>
                      ))}
                    </div>
                  )}
                </>
              )}

              {historyTab === 'video' && (
                <>
                  <div className="flex justify-end mb-2">
                    <button
                      onClick={async () => {
                        await dbService.clearVideoHistory();
                        setVideoHistory([]);
                      }}
                      className="text-[10px] text-zinc-500 hover:text-red-400 transition-colors uppercase tracking-widest"
                    >
                      {t('clearAllVideoHistory', lang)}
                    </button>
                  </div>
                  {videoHistory.length === 0 ? (
                    <div className="h-full flex flex-col items-center justify-center text-zinc-500 gap-2 opacity-50">
                      <ListVideo size={32} />
                      <p className="text-sm">{t('noVideoHistory', lang)}</p>
                    </div>
                  ) : (
                    <div className="flex flex-col gap-1">
                      {videoHistory.map((item) => (
                        <div key={item.id} className="group flex items-center gap-2 p-2 rounded-xl hover:bg-black/5 dark:hover:bg-white/5 transition-colors">
                          <div className="w-8 h-8 rounded-lg bg-orange-500/10 dark:bg-amber-500/10 flex items-center justify-center shrink-0">
                            <Play size={14} className="text-orange-500 dark:text-amber-500 ml-0.5" />
                          </div>
                          <div className="flex-1 min-w-0 cursor-pointer" onClick={() => onPlay(item.url, item.pageUrl, item.title)}>
                            <div className="text-sm text-zinc-900 dark:text-white truncate font-medium">{item.title}</div>
                            <div className="text-[10px] text-zinc-500 truncate">{item.type}</div>
                          </div>
                          <button
                            onClick={async (e) => {
                              e.stopPropagation();
                              await dbService.deleteVideoHistory(item.id);
                              setVideoHistory(prev => prev.filter(h => h.id !== item.id));
                            }}
                            className="p-1.5 text-zinc-500 hover:text-red-500 hover:bg-red-500/10 rounded-lg transition-colors opacity-0 group-hover:opacity-100"
                            title={t('delete', lang)}
                          >
                            <Trash2 size={14} />
                          </button>
                        </div>
                      ))}
                    </div>
                  )}
                </>
              )}
            </div>
          </div>
        )}

        {/* Adblock Tracker Sidebar */}
        {showAdblockSidebar && !isFullscreen && (
          <div 
            style={{ width: sidebarWidth }}
            className="flex-none bg-white/5 dark:bg-black/40 backdrop-blur-md border border-transparent dark:border-transparent rounded-2xl flex flex-col z-10 h-full overflow-hidden shadow-sm"
          >
            <div className="flex items-center justify-between h-14 px-4 bg-zinc-200/5 dark:bg-white/5 border-b border-black/10 dark:border-white/10 flex-none">
              <h2 className="text-sm font-semibold text-zinc-900 dark:text-white flex items-center gap-2">
                <ShieldCheck size={16} className="text-emerald-500" />
                <span>{lang === 'id' ? 'Pelacak Adblock' : 'Adblock Tracker'}</span>
              </h2>
              <button 
                onClick={() => setShowAdblockSidebar(false)}
                className="p-1.5 text-zinc-500 hover:text-orange-600 dark:hover:text-amber-500 hover:bg-orange-500/10 dark:hover:bg-amber-500/10 rounded-lg transition-colors"
                title="Close Tracker"
              >
                <X size={16} />
              </button>
            </div>
            
            <div className="flex-1 overflow-y-auto p-4 flex flex-col gap-4 custom-scrollbar">
              {/* Adblocker Status Card */}
              <div className="bg-white dark:bg-zinc-900/50 p-4 rounded-2xl border border-black/10 dark:border-white/5 shadow-sm flex items-center justify-between">
                <div>
                  <div className="text-xs font-bold text-zinc-500 uppercase tracking-wider">{lang === 'id' ? 'Status Pelindung' : 'Protection Status'}</div>
                  <div className="text-sm font-black mt-1 flex items-center gap-1.5">
                    <span className={`inline-block w-2.5 h-2.5 rounded-full ${adBlockEnabled ? 'bg-emerald-500' : 'bg-red-500'}`}></span>

                    {adblockStats.page[activeTabId] || 0}
                  </div>

                <div className="bg-white dark:bg-zinc-900/50 p-4 rounded-2xl border border-black/10 dark:border-white/5 shadow-sm flex flex-col justify-between h-28">
                  <div className="text-[10px] font-extrabold text-zinc-400 dark:text-zinc-500 uppercase tracking-widest leading-normal">
                    {lang === 'id' ? 'Diblokir Sejak Install' : 'Blocked Since Install'}

                    return (

                                  ipc.send('trigger-edit-password', {

'text-orange-600 dark:text-amber-500 bg-orange-50 dark:bg-amber-500/10' : 'text-zinc-700 dark:text-zinc-300 hover:bg-black/5 dark:hover:bg-white/5 hover:text-zinc-900 dark:hover:text-white'}`}

'text-orange-600 dark:text-amber-500 bg-orange-50 dark:bg-amber-500/10' : 'text-zinc-700 dark:text-zinc-300 hover:bg-black/5 dark:hover:bg-white/5 hover:text-zinc-900 dark:hover:text-white'}`}

'text-orange-600 dark:text-amber-500 bg-orange-50 dark:bg-amber-500/10' : 'text-zinc-700 dark:text-zinc-300 hover:bg-black/5 dark:hover:bg-white/5 hover:text-zinc-900 dark:hover:text-white'}`}

              <button
                onClick={handleSaveHomeAddress}
                className="px-4 py-2 text-sm font-medium bg-orange-500 hover:bg-orange-600 dark:bg-amber-500 dark:hover:bg-amber-600 text-white dark:text-black rounded-lg transition-colors"
              >
                {t('save', lang)}
              </button>
            </div>
          </div>
        </div>
      )}

      {folderPrompt.show && (
        <div className="fixed inset-0 z-[10000] bg-black/50 flex items-center justify-center p-4">
          <div className="bg-white dark:bg-[#1a1a1a] rounded-xl shadow-2xl w-full max-w-md overflow-hidden border border-black/10 dark:border-white/10">
            <div className="p-4 border-b border-black/10 dark:border-white/10">
              <h3 className="text-lg font-semibold text-zinc-900 dark:text-white">
                {folderPrompt.mode === 'create' ? t('newFolderName', lang) : t('renameFolder', lang)}
              </h3>
            </div>
            <div className="p-4">
              <input
                type="text"
                value={folderPromptInput}
                onChange={(e) => setFolderPromptInput(e.target.value)}
                onKeyDown={(e) => {
                  if (e.key === 'Enter') {
                    if (folderPrompt.mode === 'create') handleCreateFolder();
                    else handleRenameFolder();
                  }
                  if (e.key === 'Escape') setFolderPrompt({ show: false, mode: 'create' });
                }}
                className="w-full px-3 py-2 bg-black/5 dark:bg-white/5 border border-black/10 dark:border-white/10 rounded-lg text-zinc-900 dark:text-white focus:outline-none focus:border-orange-500/50 dark:focus:border-amber-500/50 transition-colors"
                placeholder={t('folderNamePlaceholder', lang)}
                autoFocus
              />
            </div>
            <div className="p-4 bg-black/5 dark:bg-white/5 flex justify-end gap-2">
              <button
                onClick={() => setFolderPrompt({ show: false, mode: 'create' })}
                className="px-4 py-2 text-sm font-medium text-zinc-700 dark:text-zinc-300 hover:bg-black/5 dark:hover:bg-white/10 rounded-lg transition-colors"
              >
                {t('cancel', lang)}
              </button>
              <button
                onClick={() => {
                  if (folderPrompt.mode === 'create') handleCreateFolder();
                  else handleRenameFolder();
                }}
                className="px-4 py-2 text-sm font-medium bg-orange-500 hover:bg-orange-600 dark:bg-amber-500 dark:hover:bg-amber-600 text-white dark:text-black rounded-lg transition-colors"
              >
                {t('save', lang)}
              </button>
            </div>
          </div>
        </div>
      )}

      {deleteConfirm.show && (
        <div className="fixed inset-0 z-[10000] bg-black/50 flex items-center justify-center p-4">
          <div className="bg-white dark:bg-[#1a1a1a] rounded-xl shadow-2xl w-full max-w-sm overflow-hidden border border-black/10 dark:border-white/10 text-center">
            <div className="p-6">
              <div className="w-12 h-12 rounded-full bg-red-100 dark:bg-red-500/20 text-red-600 dark:text-red-500 flex items-center justify-center mx-auto mb-4">
                <Trash2 size={24} />
              </div>
              <h3 className="text-lg font-semibold text-zinc-900 dark:text-white mb-2">
                {deleteConfirm.mode === 'all' 
                  ? t('deleteAllBookmarksConfirm', lang) 
                  : t('deleteFolderConfirm', lang, { name: deleteConfirm.folderName })}
              </h3>
              <p className="text-sm text-zinc-500 dark:text-zinc-400">
                {deleteConfirm.mode === 'all' 
                  ? t('deleteAllBookmarksDesc', lang) 
                  : t('deleteFolderDesc', lang)}
              </p>
            </div>
            <div className="p-4 bg-black/5 dark:bg-white/5 flex justify-end gap-2">
              <button
                onClick={() => setDeleteConfirm({ show: false, mode: 'all' })}
                className="flex-1 px-4 py-2 text-sm font-medium text-zinc-700 dark:text-zinc-300 hover:bg-black/5 dark:hover:bg-white/10 rounded-lg transition-colors"
              >
                {t('cancel', lang)}
              </button>
              <button
                onClick={() => {
                  if (deleteConfirm.mode === 'all') handleDeleteAllBookmarks();
                  else handleDeleteFolder();
                }}
                className="flex-1 px-4 py-2 text-sm font-medium text-white bg-red-600 hover:bg-red-700 rounded-lg transition-colors"
              >
                {t('delete', lang)}
              </button>
            </div>
          </div>
        </div>
      )}

      {showQuickAccessModal && (
        <div className="fixed inset-0 z-[10000] bg-black/55 backdrop-blur-md flex items-center justify-center p-4">
          <div className="bg-white dark:bg-[#111111] border border-black/10 dark:border-white/10 rounded-2xl shadow-2xl w-full max-w-md p-6 flex flex-col gap-4">
            <div className="flex justify-between items-center border-b border-black/10 dark:border-white/10 pb-3">
              <h3 className="text-zinc-900 dark:text-zinc-100 font-bold text-sm tracking-wide uppercase flex items-center gap-2">
                <Globe size={16} className="text-orange-600 dark:text-amber-500 animate-pulse" />
                {quickAccessModalMode === 'add' ? t('quickAccessAddTitle', lang) : t('quickAccessEditTitle', lang)}
              </h3>
              <button
                onClick={() => setShowQuickAccessModal(false)}
                className="text-zinc-700 dark:text-zinc-300 hover:text-orange-600 dark:hover:text-amber-500 hover:bg-orange-500/10 dark:hover:bg-amber-500/10 p-1.5 rounded-full transition-colors"
              >
                <X size={16} />
              </button>
            </div>

            <div className="flex flex-col gap-3">
              <div className="flex flex-col gap-1.5">
                <label className="text-[10px] uppercase font-bold tracking-wider text-zinc-500 dark:text-zinc-400">Nama Situs / Label</label>
                <input
                  type="text"
                  value={quickAccessName}
                  onChange={(e) => setQuickAccessName(e.target.value)}
                  className="w-full bg-black/5 dark:bg-black/50 text-zinc-900 dark:text-white text-sm p-3 rounded-xl border border-black/10 dark:border-white/10 focus:border-orange-500 dark:focus:border-amber-500 focus:outline-none font-medium"
                  placeholder={t('quickAccessNamePlaceholder', lang)}
                  autoFocus
                />
              </div>

              <div className="flex flex-col gap-1.5">
                <label className="text-[10px] uppercase font-bold tracking-wider text-zinc-500 dark:text-zinc-400">Alamat URL Situs</label>
                <input
                  type="text"
                  value={quickAccessUrl}
                  onChange={(e) => setQuickAccessUrl(e.target.value)}
                  className="w-full bg-black/5 dark:bg-black/50 text-zinc-900 dark:text-white text-sm p-3 rounded-xl border border-black/10 dark:border-white/10 focus:border-orange-500 dark:focus:border-amber-500 focus:outline-none font-bold"
                  placeholder="https://example.com"
                />
              </div>
            </div>

            <div className="flex justify-end gap-3 pt-2 border-t border-black/10 dark:border-white/10">
              <button
                onClick={() => setShowQuickAccessModal(false)}
                className="px-4 py-2.5 rounded-xl text-xs font-bold uppercase tracking-wider text-zinc-700 dark:text-zinc-300 hover:bg-black/5 dark:hover:bg-white/5 transition-colors"
              >
                {t('cancel', lang)}
              </button>
              <button
                onClick={() => {
                  if (!quickAccessName.trim() || !quickAccessUrl.trim()) {
                    alert(t('quickAccessRequiredAlert', lang));
                    return;
                  }
                  
                  let formattedUrl = quickAccessUrl.trim();
                  if (!formattedUrl.startsWith('http://') && !formattedUrl.startsWith('https://')) {
                    formattedUrl = `https://${formattedUrl}`;
                  }

                  let updated: { name: string, url: string }[] = [];
                  if (quickAccessModalMode === 'add') {
                    updated = [...quickAccess, { name: quickAccessName.trim(), url: formattedUrl }];
                  } else if (quickAccessModalMode === 'edit' && quickAccessEditIndex !== null) {
                    updated = quickAccess.map((item, idx) => 
                      idx === quickAccessEditIndex ? { name: quickAccessName.trim(), url: formattedUrl } : item
                    );
                  }

                  setQuickAccess(updated);
                  localStorage.setItem('snifferQuickAccess', JSON.stringify(updated));
                  setShowQuickAccessModal(false);
                }}
                className="bg-orange-500 dark:bg-amber-500 hover:bg-orange-400 dark:hover:bg-amber-400 text-black font-bold uppercase tracking-wider text-xs px-5 py-2.5 rounded-xl transition-colors shadow-lg shadow-orange-500/10 dark:shadow-amber-500/10"
              >
                {t('saveSite', lang)}
              </button>
            </div>
          </div>
        </div>
      )}

    </div>
  );
};
